# Isekai Store Level-Up Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the 2021 Isekai Store fun-project on Next.js 16 App Router with a cinematic dark-fantasy RPG experience (preloader → gate → shopkeeper → shop with cart/checkout), backend-free with hardcoded data.

**Architecture:** Next.js 16 App Router at repo root (`app/`, `components/`, `lib/`), replacing the old `src/` pages-router tree. `app/page.tsx` is a client orchestrator owning cart state (`useReducer`) and the vertical Swiper flow (3 slides: Gate → Welcome → Shop). Cart reducer is a pure, unit-tested function. All styling is Tailwind CSS 4 via `@theme` tokens in `globals.css` plus hand-written keyframes/utility classes.

**Tech Stack:** Next.js 16, React 19, TypeScript (strict), Tailwind CSS 4 (`@tailwindcss/postcss`), Swiper 14, react-icons, Vitest, ESLint 9 (`eslint-config-next`).

**Spec:** `docs/superpowers/specs/2026-08-13-isekaistore-levelup-design.md`

## Global Constraints

- **NO `npm run build`** — user vetoed. Verification = `npm run typecheck`, `npm run lint`, `npm run test`, and dev-server smoke checks (`npm run dev` + `curl`).
- All component files that use hooks or event handlers MUST start with `"use client";`.
- Path alias `@/*` → repo root (e.g. `@/lib/cart`, `@/public/wand.webp`).
- Palette tokens (only these, exact hex): abyss `#0a0a14`, night `#12121f`, parchment `#f5efe0`, ember `#f0b33c`, arcane `#8b5cf6`, blood `#e2483d`, arcana `#34d3b2`.
- Fonts via `next/font/google`: Cinzel (display, variable), VT323 (weight 400 only — it is NOT a variable font), Nunito (body, variable).
- Rarity palette: common `#9ca3af`, uncommon `#4ade80`, rare `#60a5fa`, epic `#a78bfa`, legendary `#fbbf24`.
- Existing public images are the identity — do not delete them: `frog.gif`, `megumin.gif`, `isekaimaou.gif`, `isekai_bg.webp`, `shop.webp`, `shopkeeper.webp`, `wand.webp`, `potion.webp`, `sword.webp`, `shield.webp`, `elixir.webp`, `scroll.webp`, `karyl404.jpg`, `favicon.ico`, `vercel.svg`.
- Commit messages follow repo style: lowercase, no prefix (e.g. `feat: ...` is NOT used — plain `make the gate glow`).
- Image imports need ambient module declarations (new project has none) — see Task 1 `types/images.d.ts`.
- Old `src/` tree and legacy configs are DELETED in Task 1.

- [ ] **Step 4: Write `lib/cart.ts`** (type-only import of Item, so vitest never loads `items.ts` and its image imports)

```ts
import type { Item } from "./items";

export type CartState = Record<number, number>;

export type CartAction =
  | { type: "add"; itemId: number }
  | { type: "remove"; itemId: number }
  | { type: "setQty"; itemId: number; qty: number }
  | { type: "clear" };

export function cartReducer(state: CartState, action: CartAction): CartState {
  switch (action.type) {
    case "add":
      return { ...state, [action.itemId]: (state[action.itemId] ?? 0) + 1 };
    case "remove": {
      if (!state[action.itemId]) return state;
      const next = { ...state };
      if (next[action.itemId] > 1) next[action.itemId] -= 1;
      else delete next[action.itemId];
      return next;
    }
    case "setQty": {
      if (action.qty <= 0) {
        const cleared = { ...state };
        delete cleared[action.itemId];
        return cleared;
      }
      return { ...state, [action.itemId]: action.qty };
    }
    case "clear":
      return {};
  }
}

export function cartCount(state: CartState): number {
  return Object.values(state).reduce((sum, q) => sum + q, 0);
}

export function cartTotal(state: CartState, items: Item[]): number {
  return Object.entries(state).reduce((sum, [id, qty]) => {
    const item = items.find((i) => i.id === Number(id));
    return sum + (item ? item.price * qty : 0);
  }, 0);
}
```

```tsx
"use client";

import { useEffect, useRef, useState } from "react";

interface TypewriterProps {
  text: string;
  speed?: number;
  startDelay?: number;
  onDone?: () => void;
  className?: string;
}

export default function Typewriter({
  text,
  speed = 45,
  startDelay = 300,
  onDone,
  className,
}: TypewriterProps) {
  const [out, setOut] = useState("");
  const doneRef = useRef(false);

  useEffect(() => {
    doneRef.current = false;
    let i = 0;
    let timer: ReturnType<typeof setTimeout>;
    const tick = () => {
      if (i < text.length) {
        i += 1;
        setOut(text.slice(0, i));
        timer = setTimeout(tick, speed);
      } else if (!doneRef.current) {
        doneRef.current = true;
        onDone?.();
      }
    };
    timer = setTimeout(tick, startDelay);
    return () => clearTimeout(timer);
  }, [text, speed, startDelay, onDone]);

  return (
    <span className={className}>
      {out}
      <span className="anim-blink">▌</span>
    </span>
  );
}
```

```tsx
"use client";

import { useCallback, useReducer, useState } from "react";
import { Swiper, SwiperSlide } from "swiper/react";
import type { Swiper as SwiperType } from "swiper";
import { A11y, Mousewheel, Pagination } from "swiper/modules";
import "swiper/css";
import "swiper/css/pagination";

import EmberField from "@/components/EmberField";
import Preloader from "@/components/Preloader";
import GateSlide from "@/components/GateSlide";
import WelcomeSlide from "@/components/WelcomeSlide";
import ShopSlide from "@/components/ShopSlide";
import { cartReducer } from "@/lib/cart";
import { ITEMS } from "@/lib/items";

export default function Home() {
  const [loaded, setLoaded] = useState(false);
  const [swiper, setSwiper] = useState<SwiperType | null>(null);
  const [thankYou, setThankYou] = useState(false);
  const [cart, dispatch] = useReducer(cartReducer, {});

  const handlePreloaderDone = useCallback(() => setLoaded(true), []);

  const handleEnterShop = useCallback(() => {
    swiper?.slideTo(2, 900);
  }, [swiper]);

  const handlePurchase = useCallback(() => {
    dispatch({ type: "clear" });
    setThankYou(true);
    swiper?.slideTo(1, 900);
  }, [swiper]);

  return (
    <main className="relative">
      <EmberField />

      {!loaded ? (
        <Preloader onDone={handlePreloaderDone} />
      ) : (
        <Swiper
          className="h-screen"
          direction="vertical"
          speed={800}
          modules={[Mousewheel, Pagination, A11y]}
          mousewheel={{ forceToAxis: true }}
          pagination={{ clickable: true }}
          onSwiper={setSwiper}
          style={{ zIndex: 20 }}
        >
          <SwiperSlide>
            <GateSlide />
          </SwiperSlide>
          <SwiperSlide>
            <WelcomeSlide thankYou={thankYou} onEnterShop={handleEnterShop} />
          </SwiperSlide>
          <SwiperSlide>
            <ShopSlide
              items={ITEMS}
              cart={cart}
              dispatch={dispatch}
              onPurchase={handlePurchase}
            />
          </SwiperSlide>
        </Swiper>
      )}
    </main>
  );
}
```
